{{- /* Inline citation as a superscript linking to a source in the list. Usage: {{< cite 1 >}} or {{< cite 1 3 >}} The first citation of each source gets id="cite-N" so the source can link back to it. */ -}} [ {{- range $i, $n := .Params -}} {{- if $i }}, {{ end -}} {{- $key := printf "cite-%v" $n -}} {{- if $.Page.Store.Get $key -}} {{ $n }} {{- else -}} {{- $.Page.Store.Set $key true -}} {{ $n }} {{- end -}} {{- end -}} ] {{- /* trailing newline trimmed */ -}}
